@Composable
fun BusinessCard(name: String, subHead: String, modifier: Modifier = Modifier) {
    Column(
        modifier = modifier,
        verticalArrangement = Arrangement.Center,
        horizontalAlignment = Alignment.CenterHorizontally
    ) {
        Text(
            text = name,
            fontSize = 42.sp,
            lineHeight = 43.sp,
            textAlign = TextAlign.Center,
            modifier = Modifier.padding(10.dp),
            color = Color.White
        )
        Text(
            text = subHead,
            fontSize = 32.sp,
            modifier = Modifier.padding(10.dp),
            color = Color.White
        )
    }
}

@Composable
fun Bgrouundimg(modifier: Modifier) {
    val image = painterResource(R.drawable.wp9388599)
    Image(
        painter = image,
        contentDescription = null,
        contentScale = ContentScale.Crop,
        modifier = modifier
    )
}

@Composable
fun BusinessCardImage(
    name: String, subHead: String, pone: String, booty: String, emall: String, modifier: Modifier = Modifier
) {
    Box(modifier) {
        Bgrouundimg(
            modifier = Modifier.fillMaxSize(),
        )
        Column(
            modifier = Modifier.fillMaxSize(),
            horizontalAlignment = Alignment.CenterHorizontally,
        ) {
            Column(
                modifier = Modifier.weight(1f),
                horizontalAlignment = Alignment.CenterHorizontally,
                verticalArrangement = Arrangement.Center
            ) {
                Image(
                    painter = painterResource(R.drawable.android_logo),
                    contentDescription = null,
                    modifier = Modifier.padding(18.dp)
                )
                BusinessCard(
                    name = name,
                    subHead = subHead,
                )
            }
            BottomTex(
                pone = pone,
                booty = booty,
                emall = emall,
                modifier = Modifier.padding(bottom = 20.dp)
            )
        }
    }
}
